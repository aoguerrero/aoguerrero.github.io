PostgreSQL CLI

Initialize Database.

initdb -D $env:PG_HOME\data --username=postgres --auth=trust -E utf8

Start Server.

pg_ctl.exe -D "{path\to}\data" -l "{path\to}\log.txt" start

Start PostgreSQL CLI.

psql -U {username}

Create Database, add UUID extension.

create database {database}
\c {database}
create extension if not exists "uuid-ossp" schema pg_catalog;

Basic commands.

# Change Password
\password

# List databases.
\l

# Use database.
\c {database}

#  List schemas.
\dn 

# List tables.
\dt

# lista tables of a schema.
\dt {schema}.*

# rename database
alter database {} rename to {};

Create a copy of a database

\c postgres;
select pg_terminate_backend(pid) from pg_stat_activity where datname = '{database}' and pid <> pg_backend_pid();
create database {database_copy} with template {database};
